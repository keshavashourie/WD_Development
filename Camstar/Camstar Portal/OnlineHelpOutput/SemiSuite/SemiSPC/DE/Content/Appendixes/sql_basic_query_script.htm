SQL-Basisabfrageskript

Dies ist ein Beispiel für die SQL-Basisabfrage ss_TrackInOutSPCBasicQuery, mit der die für einen SPC-Basisaufruf erforderlichen Opcenter Execution Semiconductor-Daten gesammelt werden. Die allgemeine Struktur einer SPC-Basisabfrage ist identisch mit der SPCDataValues-Abfrage, die in "SQL-Abfrageskript" beschrieben wird. Sie können das SQL-Basisabfrageskript wie folgt anpassen:

SELECT

DATAVALUE

, INSTANCEID

, COMMENTS

, CONTAINER

, WAFERSCRIBENUMBER

, TXNTIMESTAMP /*Added to support the X AXIS display value.*/

, COMMENTABREV /*Hold the SPC annotation category or default comment abbreviation if allowed by the chart.*/

, GROUPSEQ

, RN

, BYLOTID

, BYWAFERID

, ORIGINALVALUE

FROM

(

SELECT

DATAVALUE

, INSTANCEID

, COMMENTS

, CAST(CASE WHEN ?SUBGROUPPLOTMETHOD = -1 THEN ROW_NUMBER() OVER (ORDER BY DataSource ASC, TxnTimestamp DESC , DisplaySequence DESC) ELSE ROW_NUMBER() OVER (ORDER BY DATASOURCE ASC, GROUPID DESC, WAFERSCRIBENUMBER DESC, DISPLAYSEQUENCE DESC) END AS INT) RN /*Adjusts the order of the data points based on the configured SUBGROUP plot method.*/

, CONTAINER

, WAFERSCRIBENUMBER

, TXNTIMESTAMP

, CASE WHEN SS_SPCANNOTATIONCATEGORYNAME IS NULL THEN '[i]' ELSE SS_SPCANNOTATIONCATEGORYNAME END AS COMMENTABREV

, CASE WHEN DATASOURCE = 1 THEN 'ffffffffffffffff' ELSE GROUPID END ?ConcatSymbol '/' ?ConcatSymbol GROUPID ?ConcatSymbol '/' ?ConcatSymbol CONTAINERID BYLOTID /*Format used to create the appropriate grouping for "BY LOT" processing.*/

, CASE WHEN DATASOURCE = 1 THEN 'ffffffffffffffff' ELSE GROUPID END ?ConcatSymbol '/' ?ConcatSymbol CONTAINERID ?ConcatSymbol '/' ?ConcatSymbol WAFERSCRIBENUMBER BYWAFERID /*Format used to create the appropriate grouping for "BY WAFER" processing.*/

, CAST(DENSE_RANK() OVER (ORDER BY GROUPID) AS INT) GROUPSEQ /*Allows the Macro to analyze the Last Group if configured on the SPC rule data point analysis method.*/

/*

*/

, ORIGINALVALUE

FROM

(

SELECT

CASE WHEN STDP.OverrideValue IS NULL THEN CAST(WLDD.WIPDataValue AS FLOAT) ELSE STDP.OverrideValue END DATAVALUE

, WLDD.WIPDataValue ORIGINALVALUE

, WLDD.WIPLotDetailsDataId INSTANCEID

, '' COMMENTS

, WLDD.TxnTimestamp

, WLDD.DisplaySequence

, WLDD.WaferScribeNumber

, 1 DataSource

, C.CONTAINERNAME CONTAINER

, WLDD.WIPLOTDETAILSID GROUPID

, STDP.SS_SPCANNOTATIONCATEGORYNAME

FROM

Container C

INNER JOIN A_WIPLotDetails WLD ON C.CurrentWIPLotId = WLD.WIPLotId

INNER JOIN A_WIPLotDetailsData WLDD ON WLD.WIPLotDetailsId = WLDD.WIPLotDetailsId

LEFT OUTER JOIN A_WIPEquipment WE ON WLD.WIPLotDetailsId = WE.WIPLotDetailsLinkId

LEFT OUTER JOIN RESOURCEDEF RD ON WLDD.EQUIPMENTID = RD.RESOURCEID

INNER JOIN A_WIPDataName WDN ON WLDD.WIPDataNameId = WDN.WIPDataNameId

LEFT OUTER JOIN A_SPCTxnDataPoint STDP ON WLDD.WIPLotDetailsDataId= STDP.SPCTxnDataPointName

WHERE

WDN.WIPDataNameName LIKE ?WIPDATANAME

AND WLDD.WIPDataValue IS NOT NULL

AND WDN.FieldType = 'NUMBER'

AND WLDD.ServiceName LIKE ?ServiceName

AND ((?ServiceName <> 'TrackInLot' AND WE.TrackStatus = 'INPROCESS' AND WE.EquipmentName LIKE ?EQUIPMENT) OR (?ServiceName = 'TrackInLot' AND RD.ResourceName LIKE ?EQUIPMENT))

AND C.ContainerName = ?CONTAINER

UNION ALL

SELECT

CASE WHEN STDP.OverrideValue IS NULL THEN CAST(WEDH.WIPDataValue AS FLOAT) ELSE STDP.OverrideValue END DATAVALUE

, WEDH.WIPDataValue ORIGINALDATAVALUE

, CASE WHEN WEDH.SS_WIPLOTDETAILSDATAID IS NULL THEN WIPEquipmentDataHistoryId ELSE WEDH.SS_WIPLOTDETAILSDATAID END INSTANCEID

, STDP.Description COMMENTS

, WEDH.TxnTimestamp

/*

*/

, WEDH.DisplaySequence

, WEDH.WaferScribeNumber

, 2 DataSource

, C.CONTAINERNAME CONTAINER

, C.CONTAINERID CONTAINERID

, WEDH.WIPEQUIPMENTHISTORYID GROUPID

, STDP.SS_SPCANNOTATIONCATEGORYNAME

FROM

Container C

INNER JOIN A_WIPEquipmentHistory WE ON C.ContainerId = WE.ContainerId

INNER JOIN A_WIPEquipmentDataHistory WEDH ON WE.WIPEquipmentHistoryId = WEDH.WIPEquipmentHistoryId

INNER JOIN A_WIPDataName WDN ON WEDH.WIPDataNameId = WDN.WIPDataNameId

LEFT OUTER JOIN A_SPCTxnDataPoint STDP ON (WEDH.SS_WIPLOTDETAILSDATAID = STDP.SPCTxnDataPointName OR WEDH.WIPEQUIPMENTDATAHISTORYID = STDP.SPCTxnDataPointName)

INNER JOIN PRODUCT P ON C.ProductID = P.ProductID

WHERE

WDN.WIPDataNameName LIKE ?WIPDATANAME

AND WEDH.WIPDataValue IS NOT NULL

AND WEDH.ServiceName LIKE ?ServiceName

AND WDN.FieldType = 'NUMBER'

AND WE.EquipmentName LIKE ?EQUIPMENT

AND ((?SPCMatrixID <> '%' AND WE.SPCMATRIXID LIKE ?SPCMatrixID) OR (?SPCMatrixID = '%'))

AND (CASE WHEN STDP.ExcludeData IS NULL THEN 0 ELSE STDP.ExcludeData END) = 0

?CustomFilters /*Utilized when working with the SPC Matrix. All custom filters will be applied to the historical SPC results.*/

) Query1

WHERE

(?SUBGROUPPLOTMETHOD = 0 ) OR (?SUBGROUPPLOTMETHOD = 1 AND WAFERSCRIBENUMBER IS NOT NULL) OR (?SUBGROUPPLOTMETHOD = -1)

) Query2

WHERE RN <= ?ResultSetSizeLimit

ORDER BY RN DESC
