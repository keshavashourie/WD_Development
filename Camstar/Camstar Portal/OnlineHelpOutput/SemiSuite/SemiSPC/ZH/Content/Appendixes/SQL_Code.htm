SQL 查询脚本

这是 SQL 查询的示例，用于收集 SPC 调用所需的 Opcenter Execution Semiconductor 数据。有关此查询的其他重要信息，注意查看签入式注释，并参考“SQL 查询脚本注释”。

SELECT

DATAVALUE /* Numeric value to be plotted on chart based on chart type. */

, INSTANCEID /* Unique Id of each set of data. Required to show comments or exclusions.*/

, COMMENTS  /* Historical comments. */

, COMMENTABREV

, ORIGINALVALUE

FROM

(

SELECT

DATAVALUE

, INSTANCEID

, COMMENTS

, ROW_NUMBER() OVER (ORDER BY DataSource DESC, TxnTimestamp ASC, DisplaySequence DESC) RN  /* Clause must remain as is with the exception that you can add "Waferscribenumber ASC" as explained in Notes.*/

,CASE WHEN SS_SPCANNOTATIONCATEGORYNAME IS NULL THEN '[i]' ELSE SS_SPCANNOTATIONCATEGORYNAME END AS COMMENTABREV

, ORIGINALVALUE

FROM /* Return field names must be the same for both sub-queries below. */

 ( SELECT /* Get WIP processing data on current lot */

CASE WHEN STDP.OverrideValue IS NULL THEN CAST(WLDD.WIPDataValue AS FLOAT) ELSE STDP.OverrideValue END DATAVALUE

, CAST(WLDD.WIPDataValue AS FLOAT) ORIGINALVALUE

, WLDD.WIPLotDetailsDataId INSTANCEID

, '' COMMENTS

, WLDD.TxnTimestamp

, WLDD.DisplaySequence

, 1 DataSource

, STDP.SS_SPCANNOTATIONCATEGORYNAME

FROM

Container C

INNER JOIN A_WIPLotDetails WLD ON C.CurrentWIPLotId = WLD.WIPLotId

INNER JOIN A_WIPLotDetailsData WLDD ON WLD.WIPLotDetailsId = WLDD.WIPLotDetailsId

INNER JOIN A_WIPEquipment WE ON WLD.WIPLotDetailsId = WE.WIPLotDetailsLinkId /* Allows SPC to be evaluated by equipment*/

INNER JOIN A_WIPDataName WDN ON WLDD.WIPDataNameId = WDN.WIPDataNameId /*Allows retrieval of WIP Data specified in query parameters */

/*

*/

LEFT OUTER JOIN A_SPCTxnDataPoint STDP ON WLDD.WIPLotDetailsDataId= STDP.SPCTxnDataPointName /*Allows for retrieval of any overridden values on current lot.*/

WHERE

WDN.WIPDataNameName LIKE ?WIPDATANAME /*Maps to the WIP Data query parameter */

AND WLDD.WIPDataValue IS NOT NULL

AND WDN.FieldType = 'NUMBER'

AND WLDD.ServiceName = 'TrackOutLot' /* Change as needed to represent the service*/

AND WE.TrackStatus = 'INPROCESS' /* Remains INPROCESS as long as query is retrieving data from a Lot that is still on a piece of equipment. */

AND WE.EquipmentName = ?EQUIPMENT /* Maps to equipment query parameter */

AND C.ContainerName = ?CONTAINER /* Maps to container query parameter */

UNION ALL /* Add historical data */

SELECT

CASE WHEN STDP.OverrideValue IS NULL THEN CAST(WEDH.WIPDataValue AS FLOAT) ELSE STDP.OverrideValue END DATAVALUE

, CAST(WEDH.WIPDataValue AS FLOAT) ORIGINALVALUE

, WEDH.WIPEquipmentDataHistoryId INSTANCEID

, STDP.Description COMMENTS

, WEDH.TxnTimestamp

, WEDH.DisplaySequence

, 2 DataSource

, STDP.SS_SPCANNOTATIONCATEGORYNAME

FROM

Container C

INNER JOIN A_WIPEquipmentHistory WE ON C.ContainerId = WE.ContainerId

INNER JOIN A_WIPEquipmentDataHistory WEDH ON WE.WIPEquipmentHistoryId = WEDH.WIPEquipmentHistoryId

INNER JOIN A_WIPDataName WDN ON WEDH.WIPDataNameId = WDN.WIPDataNameId

LEFT OUTER JOIN A_SPCTxnDataPoint STDP ON (WEDH.SS_WIPLOTDETAILSDATAID IS NOT NULL AND WEDH.SS_WIPLOTDETAILSDATAID = STDP.SPCTxnDataPointName) OR ( WEDH.SS_WIPLOTDETAILSDATAID IS NULL AND WEDH.WIPEquipmentDataHistoryId = STDP.SPCTxnDataPointName)

WHERE

WDN.WIPDataNameName LIKE ?WIPDATANAME

AND WEDH.WIPDataValue IS NOT NULL

AND WEDH.ServiceName = 'TrackOutLot' /*If the same WIP Data name is being collected during TrackIn and TrackOut, specify which transaction. */

AND WDN.FieldType = 'NUMBER'

AND WE.EquipmentName = ?EQUIPMENT

AND (CASE WHEN STDP.ExcludeData IS NULL THEN 0 ELSE STDP.ExcludeData END) = 0 /* Exclude data points that were previously marked for exclusion by end user. */

) Query1

) Query2

WHERE

RN <= 200 /* Ensures that no more than 200 rows are returned. Refer to "SQL 查询脚本注释" for details.*/
